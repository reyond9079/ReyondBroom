cmake_minimum_required(VERSION 3.20)
project(ReyondCompileUI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 COMPONENTS Widgets Svg REQUIRED)
set(tb ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/trenchbroom)
set(ui ${tb}/TbUiLib)
file(GLOB controls ${ui}/src/*.cpp ${ui}/include/ui/*.h)
add_library(reyond-compile-ui SHARED ${controls} compile_ui.cpp resources.cpp api.h)
target_include_directories(reyond-compile-ui PRIVATE ${ui}/include ${tb}/KdLib/include ${tb}/TbBaseLib/include ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_link_libraries(reyond-compile-ui PRIVATE Qt6::Widgets Qt6::Svg)
# No model/EL dependency in this UI: the host executes every operation through JSON RPC.
target_sources(reyond-compile-ui PRIVATE ${tb}/KdLib/src/contracts.cpp)
if(MSVC)
 target_compile_options(reyond-compile-ui PRIVATE /utf-8 /EHsc)
endif()
set_target_properties(reyond-compile-ui PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")

if(UNIX AND NOT APPLE)
 target_link_options(reyond-compile-ui PRIVATE -Wl,-z,defs)
endif()
