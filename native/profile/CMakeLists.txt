cmake_minimum_required(VERSION 3.20)
project(ReyondCompilationCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(tb ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/trenchbroom)
file(GLOB el ${tb}/TbElLib/src/*.cpp)
file(GLOB kd ${tb}/KdLib/src/*.cpp)
file(GLOB base ${tb}/TbBaseLib/src/*.cpp)
file(GLOB fs ${tb}/TbFsLib/src/*.cpp)
add_library(reyond-profile STATIC ${el} ${kd} ${base} ${fs} el.cpp files.cpp export.cpp)
target_include_directories(reyond-profile PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/.. PRIVATE ${tb}/TbElLib/include ${tb}/TbBaseLib/include ${tb}/KdLib/include ${tb}/TbFsLib/include ${tb}/VmLib/include ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fmt)
target_compile_definitions(reyond-profile PRIVATE FMT_HEADER_ONLY)
# GCC 16/MinGW no longer supplies cstdint transitively to the upstream VM headers.
target_precompile_headers(reyond-profile PRIVATE <cstdint>)
if(MSVC)
 target_compile_options(reyond-profile PRIVATE /utf-8 /EHsc)
endif()
option(RB_PROFILE_TESTS "Build profile compatibility and upstream EL tests" OFF)
if(RB_PROFILE_TESTS)
 enable_testing()
 include(FetchContent)
 FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG 2b60af89e23d28eefc081bc930831ee9d45ea58b)
 FetchContent_MakeAvailable(Catch2)
 file(GLOB eltests ${tb}/TbElLib/test/src/*.cpp)
 add_executable(trenchbroom-el-tests ${eltests} ${tb}/TbElLib/test-utils/src/TestUtils.cpp)
 target_include_directories(trenchbroom-el-tests PRIVATE ${tb}/TbElLib/include ${tb}/TbBaseLib/include ${tb}/KdLib/include ${tb}/TbElLib/test-utils/include ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fmt)
 target_compile_definitions(trenchbroom-el-tests PRIVATE FMT_HEADER_ONLY CATCH_CONFIG_ENABLE_ALL_STRINGMAKERS=1)
 target_link_libraries(trenchbroom-el-tests PRIVATE reyond-profile Catch2::Catch2WithMain)
 add_test(NAME upstream-el-corpus COMMAND trenchbroom-el-tests)
 add_executable(compilation-profile-import-test ../../tests/trenchbroom-profiles.cpp)
 set_target_properties(compilation-profile-import-test PROPERTIES CXX_STANDARD 17)
 target_link_libraries(compilation-profile-import-test PRIVATE reyond-profile)
 add_test(NAME profile-import COMMAND compilation-profile-import-test ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/CompilationProfiles-cpu-count.cfg)
 add_executable(compilation-profile-core-test ../../tests/compilation-profile-core.cpp)
 target_link_libraries(compilation-profile-core-test PRIVATE reyond-profile)
 add_test(NAME profile-core COMMAND compilation-profile-core-test)
 # Assertions in these tests are part of the oracle, also in Release builds.
 foreach(target trenchbroom-el-tests compilation-profile-import-test compilation-profile-core-test)
  if(MSVC)
   target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
   target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
 endforeach()
endif()
if(RB_PROFILE_TESTS)
 add_executable(compilation-profile-probe ../../tests/compilation-profile-probe.cpp)
 if(MINGW)
  target_link_options(compilation-profile-probe PRIVATE -municode -static)
 endif()
endif()
